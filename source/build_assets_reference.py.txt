def extract(im,r,c):
    w,h=im.size
    cell=im.crop((round(c*w/4),round(r*h/4),round((c+1)*w/4),round((r+1)*h/4)))
    a=np.array(cell)
    mask=a[:,:,3]>100
    labels,n=ndimage.label(mask)
    sizes=np.bincount(labels.ravel())
    keep=sizes>=max(12,int(mask.sum()*.002)); keep[0]=False
    good=keep[labels]
    # Remove detached source speckles only; preserve original RGB and antialiasing.
    near=ndimage.binary_dilation(good,iterations=2)
    a[:,:,3]=np.where(near,a[:,:,3],0)
    cell=Image.fromarray(a)
    box=Image.fromarray((good*255).astype('uint8')).getbbox()
    assert box, (r,c)
    return cell.crop(box)
def fit(im,size=128,extent=112,scale=None,bottom=True):
    scale=scale or min(extent/im.width,extent/im.height)
    ni=im.resize((max(1,round(im.width*scale)),max(1,round(im.height*scale))),Image.Resampling.NEAREST)
    frame=Image.new('RGBA',(size,size))
    frame.alpha_composite(ni,((size-ni.width)//2,size-6-ni.height if bottom else (size-ni.height)//2))
    return frame

# Short original synthesized sound effects. No external recordings or libraries.
RATE=22050
def sound(name,notes,duration=.25,noise=0):
    n=int(RATE*duration); rng=np.random.default_rng(47)
    t=np.arange(n)/RATE; signal=np.zeros(n)
    chunks=np.array_split(np.arange(n),len(notes))
    for ids,freq in zip(chunks,notes):
        u=np.arange(len(ids))/RATE
        env=np.sin(np.linspace(0,np.pi,len(ids)))**1.5
        signal[ids]=(.7*np.sin(2*np.pi*freq*u)+.18*np.sin(4*np.pi*freq*u)+noise*rng.uniform(-1,1,len(ids)))*env
    signal*=.28/max(1,np.max(np.abs(signal)))
    p=ART/'audio'/(name+'.wav')
    with wave.open(str(p),'wb') as w:
        w.setnchannels(1);w.setsampwidth(2);w.setframerate(RATE);w.writeframes((signal*32767).astype('<i2').tobytes())
    record(p,kind='audio',sample_rate=RATE,channels=1,duration=duration,origin='original procedural synthesis')
